        import java.util.ArrayList;

        public class GradeList
        {
        private ArrayList grades;

        public GradeList()
        {
            this.grades = new ArrayList<>();
        }

        public int size()
        {
            return grades.size();
        }

        public void addGrade(Grade grade)
        {
            grades.add(grade);
        }

        public void addGrade(int index, Grade grade)
        {
            grades.add(index, grade);
        }

        public void removeGrade(int index)
        {
            grades.remove(index);
        }

        public Grade getGrade(int index)
        {
            return grades.get(index);
        }

        public int getMinimumGrade()
        {
            int min = grades.getFirst().getGrade();
            for (Grade g : grades)
            {
            if (g.getGrade() < min)
            {
                min = g.getGrade();
            }
            }
            return min;
        }

        public int getMaximumGrade()
        {
            int max = grades.getFirst().getGrade();
            for (Grade g : grades)
            {
            if (g.getGrade() > max)
            {
                max = g.getGrade();
            }
            }
            return max;
        }

        public double getAverageGrade()
        {
            int total = grades.getFirst().getGrade();
            for (Grade g: grades)
            {
            total += g.getGrade();
            }
            return (double) total / grades.size();
        }

        @Override
        public String toString()
        {
            String result = "";
            for (Grade g : grades)
            {
            result += g + "\n";
            }
            return result;
        }

        @Override
        public boolean equals(Object obj)
        {
            if (this == obj) return true;
            if (obj == null || getClass() != obj.getClass()) return false;
            GradeList other = (GradeList) obj;
            return grades.equals(other.grades);
        }
        }

    import java.io.*;

    public class TextFileGradeListPersistence implements GradeListPersistence
    {

    @Override public GradeList load(String fileName) throws PersistenceException
    {
        GradeList gradelist = new GradeList();
        try (BufferedReader in = new BufferedReader(new FileReader(fileName)))
        {
        String line = in.readLine();
        while (line != null)
        {
            String[] parts = line.split(";", 2);
            String course = parts[0];
            int gradevalue = Integer.parseInt(parts[1]);

            Grade grade = new Grade(gradevalue,course);
            gradelist.addGrade(grade);

            line = in.readLine();
        }
        return gradelist;
        }
        catch (IOException | RuntimeException e)
        {
        throw new PersistenceException("Could not load notebook", e);
        }

    }

    @Override public void save(String fileName, GradeList gradeList)
        throws PersistenceException
    {
        try (PrintWriter out = new PrintWriter(fileName))
        {
        for (int i = 0; i < gradeList.size(); i++)
        {
            String course = gradeList.getGrade(i).getCourse();
            int grade = gradeList.getGrade(i).getGrade();
            out.println(course + ";" + grade);
        }
        }
        catch (IOException e)
        {
        throw new PersistenceException("Could not save notebook", e);
        }

    }
    }
